cmake_minimum_required(VERSION 3.20)
project(c6_example_peer)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(c6_example_peer main.cpp)
target_compile_features(c6_example_peer PRIVATE cxx_std_17)
target_compile_options(c6_example_peer PRIVATE -Wall -Wextra -Werror)
target_link_libraries(c6_example_peer PRIVATE ${std_msgs_TARGETS} rclcpp::rclcpp)
install(TARGETS c6_example_peer DESTINATION lib/${PROJECT_NAME})
ament_package()
